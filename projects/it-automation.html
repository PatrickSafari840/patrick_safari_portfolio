#!/usr/bin/env python3
"""Resize, rotate and convert every image in a folder (Pillow)."""
import os
from PIL import Image

SRC, DST = "images/", "/opt/icons/"
os.makedirs(DST, exist_ok=True)

for name in os.listdir(SRC):
    if name.startswith("."):
        continue
    with Image.open(os.path.join(SRC, name)) as im:
        im = im.rotate(-90).resize((128, 128)).convert("RGB")
        im.save(os.path.join(DST, os.path.splitext(name)[0] + ".jpeg"), "JPEG")
        print(f"converted {name}")
import os

def rename_files(directory, prefix):
    """Give every file in a folder a clean, numbered name."""
    files = sorted(f for f in os.listdir(directory) if os.path.isfile(os.path.join(directory, f)))
    for count, filename in enumerate(files, start=1):
        ext = os.path.splitext(filename)[1].lower()
        new_name = f"{prefix}_{count:03d}{ext}"
        os.rename(os.path.join(directory, filename), os.path.join(directory, new_name))
        print(f"{filename} -> {new_name}")
#!/usr/bin/env python3
"""Alert when the server is running low on resources."""
import shutil, socket, psutil

def check():
    problems = []
    if psutil.cpu_percent(interval=1) > 80:
        problems.append("CPU usage is over 80%")
    if shutil.disk_usage("/").free / shutil.disk_usage("/").total < 0.2:
        problems.append("Available disk space is less than 20%")
    if psutil.virtual_memory().available < 500 * 1024 * 1024:
        problems.append("Available memory is less than 500MB")
    if socket.gethostbyname("localhost") != "127.0.0.1":
        problems.append("localhost cannot be resolved to 127.0.0.1")
    return problems

for p in check():
    print("Error -", p)   # in production this sends an email alert
