    The same logic in PythonLa même logique en Python

    How I would run the log rule on a server, plus the SQL query I would use to pull failed logins from a database of login attempts.Comment j'exécuterais la règle sur un serveur, et la requête SQL pour extraire les échecs d'une base de tentatives de connexion.

    import re
from collections import defaultdict, deque
from datetime import datetime

PATTERN = re.compile(r"^(\w+\s+\d+\s[\d:]+).*Failed password for (?:invalid user )?(\S+) from ([\d.]+)")
THRESHOLD, WINDOW = 5, 60  # failures within 60 seconds

def detect(lines, year=2024):
    windows = defaultdict(deque)
    alerts = {}
    for line in lines:
        m = PATTERN.search(line)
        if not m:
            continue
        ts = datetime.strptime(f"{year} {m[1]}", "%Y %b %d %H:%M:%S")
        ip, q = m[3], windows[m[3]]
        q.append(ts)
        while (ts - q[0]).total_seconds() > WINDOW:
            q.popleft()
        if len(q) >= THRESHOLD:
            alerts[ip] = max(alerts.get(ip, 0), len(q))
    return alerts

with open("/var/log/auth.log") as f:
    for ip, peak in detect(f).items():
        print(f"ALERT {ip}: {peak} failures/60s -> sudo ufw deny from {ip}")
    -- Failed logins after business hours, grouped by source
SELECT  source_ip,
        COUNT(*)                    AS failures,
        COUNT(DISTINCT username)    AS users_tried,
        MIN(login_time)             AS first_seen,
        MAX(login_time)             AS last_seen
FROM    log_in_attempts
WHERE   success = FALSE
  AND   login_time > '18:00'
GROUP BY source_ip
HAVING  COUNT(*) >= 5
ORDER BY failures DESC;
    # Remove IP addresses that lost access from the allow list (Google Cybersecurity lab)
import_file = "allow_list.txt"
remove_list = ["192.168.97.225", "192.168.158.170", "192.168.201.40"]

with open(import_file, "r") as file:
    ip_addresses = file.read().split()

ip_addresses = [ip for ip in ip_addresses if ip not in remove_list]

with open(import_file, "w") as file:
    file.write("\n".join(ip_addresses))
