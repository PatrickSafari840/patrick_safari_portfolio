    CREATE TABLE tasks (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    title       VARCHAR(80)  NOT NULL,
    priority    ENUM('high','med','low') NOT NULL DEFAULT 'med',
    completed   BOOLEAN      NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_completed (completed)
);
    <?php
header('Content-Type: application/json');
$pdo = new PDO('mysql:host=localhost;dbname=todo;charset=utf8mb4', getenv('DB_USER'), getenv('DB_PASS'),
               [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]);
$id = isset($_GET['id']) ? (int) $_GET['id'] : null;
$body = json_decode(file_get_contents('php://input'), true) ?? [];

switch ($_SERVER['REQUEST_METHOD']) {
    case 'GET':
        echo json_encode($pdo->query('SELECT * FROM tasks ORDER BY created_at DESC')->fetchAll(PDO::FETCH_ASSOC));
        break;
    case 'POST':
        $title = trim($body['title'] ?? '');
        if ($title === '' || mb_strlen($title) > 80) { http_response_code(422); echo json_encode(['error' => 'Invalid title']); exit; }
        $stmt = $pdo->prepare('INSERT INTO tasks (title, priority) VALUES (?, ?)');
        $stmt->execute([$title, $body['priority'] ?? 'med']);
        http_response_code(201);
        echo json_encode(['id' => (int) $pdo->lastInsertId()]);
        break;
    case 'PATCH':
        $stmt = $pdo->prepare('UPDATE tasks SET completed = ? WHERE id = ?');
        $stmt->execute([(bool) ($body['completed'] ?? false), $id]);
        echo json_encode(['updated' => $stmt->rowCount()]);
        break;
    case 'DELETE':
        $pdo->prepare('DELETE FROM tasks WHERE id = ?')->execute([$id]);
        http_response_code(204);
        break;
    default:
        http_response_code(405);
}
    async function api(method, path, body) {
  const res = await fetch(`/api/tasks.php${path}`, {
    method,
    headers: { 'Content-Type': 'application/json' },
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
  return res.status === 204 ? null : res.json();
}

async function toggle(task) {
  task.completed = !task.completed;         // optimistic update
  render();
  try {
    await api('PATCH', `?id=${task.id}`, { completed: task.completed });
  } catch (err) {
    task.completed = !task.completed;       // roll back on failure
    render();
    showToast('Could not save. Please try again.');
  }
}
