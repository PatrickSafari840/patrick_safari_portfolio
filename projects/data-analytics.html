Why it matters for businessPourquoi c'est important pour une entreprise

A firm that sells in francs but buys inputs in dollars sees its margin shrink every year. Pricing in dollars, holding dollar reserves or indexing contracts are the usual hedges.Une entreprise qui vend en francs mais achète en dollars voit sa marge fondre chaque année. Prix en dollars, réserves en devises ou contrats indexés sont les protections habituelles.

import requests, pandas as pd

COUNTRIES = "COD;UGA;KEN;EMU;GBR;CHN;IND;JPN"
URL = (f"https://api.worldbank.org/v2/country/{COUNTRIES}"
       "/indicator/PA.NUS.FCRF?format=json&date=2012:2024&per_page=500")

meta, rows = requests.get(URL, timeout=30).json()
df = pd.DataFrame({
    "iso3":  [r["countryiso3code"] for r in rows],
    "year":  [int(r["date"]) for r in rows],
    "rate":  [r["value"] for r in rows],          # local currency units per USD
}).dropna()

df.to_csv("fx_rates.csv", index=False)
print(df.pivot(index="year", columns="iso3", values="rate").tail())
-- Index each currency to its 2012 value and compute year-over-year change
WITH base AS (
    SELECT iso3, rate AS rate_2012
    FROM fx_rates
    WHERE year = 2012
)
SELECT  f.iso3,
        f.year,
        f.rate,
        ROUND(100.0 * f.rate / b.rate_2012, 1)                                   AS index_2012,
        ROUND(100.0 * (f.rate / LAG(f.rate) OVER (PARTITION BY f.iso3 ORDER BY f.year) - 1), 1) AS yoy_pct
FROM    fx_rates f
JOIN    base b USING (iso3)
ORDER BY f.iso3, f.year;
import pandas as pd

df = pd.read_csv("fx_rates.csv").pivot(index="year", columns="iso3", values="rate")
yoy = df.pct_change() * 100

summary = pd.DataFrame({
    "change_2012_2023_%": (df.loc[2023] / df.loc[2012] - 1) * 100,
    "avg_yearly_%":       ((df.loc[2023] / df.loc[2012]) ** (1 / 11) - 1) * 100,
    "volatility_std_%":   yoy.std(),
    "worst_year":         yoy.idxmax(),
}).round(1).sort_values("change_2012_2023_%", ascending=False)

print(summary)
